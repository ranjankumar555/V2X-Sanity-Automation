# Establish SSH connection
def ssh_command(command):
    """Runs an SSH command and returns the output."""
    try:
        # Create SSH client
        client = paramiko.SSHClient()
        client.set_missing_host_key_policy(paramiko.AutoAddPolicy())  # Automatically add host key
       
        # Load the private key for authentication
        private_key = paramiko.RSAKey.from_private_key_file(private_key_path)
       
        # Connect to the server using the private key
        client.connect(hostname, username=username, pkey=private_key, look_for_keys=False, allow_agent=False, banner_timeout=200, timeout=30)
       
        # Execute the command
        stdin, stdout, stderr = client.exec_command(command)
       
        # Automatically send "Enter" after the command if needed
        stdin.write("\n")
        stdin.flush()
       
        output = stdout.read().decode().strip()
       
        # Close SSH connection
        client.close()
       
        return output
    except Exception as e:
        print(f"Error running command over SSH: {e}")
        return None
 
# Establish SFTP connection to transfer files
# New function to push file content via SSH using 'cat >'
def sftp_transfer(local_file, remote_path):
    """Transfers files using SFTP."""
    try:
        # Create SFTP client
        transport = paramiko.Transport((hostname, 22))
        transport.connect(username=username, pkey=paramiko.RSAKey.from_private_key_file(private_key_path))
       
        sftp = paramiko.SFTPClient.from_transport(transport)
       
        # Transfer the file
        sftp.put(local_file, remote_path)
       
        # Close SFTP session
        sftp.close()
        transport.close()
       
        print(f"Successfully transferred {local_file} to {remote_path}")
    except Exception as e:
        print(f"Error during SFTP transfer: {e}")
